{{- /* Obsidian-friendly images. - Paths relative to the vault (static/images/x.jpg) are served from /images/x.jpg. - The alt text can start with a position: left, right, center, wide. Anything after a "|" is the caption: ![right|Sunset at Suta](static/images/x.jpg) - Photos on consecutive lines stay inline so CSS can lay them out as a row. */ -}} {{- $src := .Destination -}} {{- if hasPrefix $src "static/" }}{{ $src = strings.TrimPrefix "static" $src }}{{ end -}} {{- $parts := split .PlainText "|" -}} {{- $first := index $parts 0 | strings.TrimSpace | lower -}} {{- $pos := "" -}} {{- $caption := .Title -}} {{- $alt := .PlainText -}} {{- if in (slice "left" "right" "center" "wide") $first -}} {{- $pos = $first -}} {{- $alt = "" -}} {{- if gt (len $parts) 1 }}{{ $caption = $caption | default (delimit (after 1 $parts) "|" | strings.TrimSpace) }}{{ $alt = $caption }}{{ end -}} {{- end -}} {{- if .IsBlock -}} {{- else -}} {{- end -}}
